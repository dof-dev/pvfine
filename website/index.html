package main

import (
    "fmt"
    "log"
    "pvfine/internal/pvf"
)

func main() {
    // 1. 打开归档 (自动识别 90US / 90CN / 110US)
    archive, err := pvf.Open("Script.pvf")
    if err != nil {
        log.Fatalf("打开失败: %v", err)
    }

    // 2. 毫秒级根据路径索引条目
    idx, found := archive.Find("equipment/common/1003.equ")
    if found {
        text, _ := archive.Text(idx)
        fmt.Println("原脚本:", text)

        // 3. 内存 Overlay 修改
        archive.SetText(idx, text + "\n# pvfine mod\n")
    }

    // 4. 原子重打包并另存为
    archive.SaveAs("Script_modified.pvf")
    fmt.Println("保存成功！")
}
